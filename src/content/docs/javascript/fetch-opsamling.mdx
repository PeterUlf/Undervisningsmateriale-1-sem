import Slide from "../../../components/Slide.astro";
import { Code } from "astro/components";

export const hintTransformer = {
  name: "fetch-hint-transformer",
  span(hast, _line, _col, _lineElement, token) {
    if (typeof token.content === "string" && token.content.includes("HVAD")) {
      this.addClassToHast(hast, "fetch-hint");
    }
  },
};

export const codeWithHints = String.raw`const productURL = "HVAD er denne URL??";
const productcontainer = "HVAD er denne container, og hvor finder du den?";
function getData() {
fetch(productURL).then((res) => res.json().then((data) => show(data)));
}
function show(data) {
productcontainer.innerHTML = \`
HVAD bør der ske her og forklar første og anden del af \${Hvad.Hvorfor}.....
\`;
}
getData(); 
// HVAD sker der her?`;
